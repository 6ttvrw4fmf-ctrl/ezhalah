-- ⚡ 2026-10-16 (backlog 576): the dangling-run remediation could never fix what its detector raised.
-- mon_detect_dangling_scrape_run() calls a row dangling 3 h after it started (dwelleo 6 h) and
-- verify_dangling_scrape_run() checks the same window, but remediate_dangling_scrape_run() only ran
-- mon_reconcile_dangling_scrape_runs(), which closes rows older than 12 h. So every dangling row
-- spent its 3 attempts closing nothing and escalated to a P0 remediation_exhausted (muktamel shard
-- 1/8, 10-16 03:46 UTC, lost runner: attempts 07:15/08:00/08:45, verified=false each time).
-- The remediation now closes THIS platform's rows at the detector's own threshold. Safe because
-- db.end_run() updates its row by id unconditionally: a run still alive would overwrite this.
create or replace function public.remediate_dangling_scrape_run(p_dedup text, p_platform text)
 returns void
 language plpgsql
 security definer
 set search_path to 'public'
as $function$
begin
  perform public.mon_reconcile_dangling_scrape_runs();
  update public.scrape_runs
     set finished_at = now(),
         ok = false,
         notes = 'orphaned — presumed killed (SIGKILL/OOM/lost runner) before end_run(); closed by '
                 'remediate_dangling_scrape_run() at the dangling detector''s own threshold. '
                 'rows_seen/rows_upserted are NOT meaningful for this row.'
   where platform = p_platform
     and ok is null
     and finished_at is null
     and started_at < now() - (case when p_platform = 'dwelleo' then interval '6 hours' else interval '3 hours' end);
end $function$;

-- check: a row the detector would raise (4 h old) is closed by one remediation and then verifies.
do $check$
begin
  begin
    insert into public.scrape_runs (platform, started_at) values ('zz_selftest_dangling', now() - interval '4 hours');
    perform public.remediate_dangling_scrape_run('selftest', 'zz_selftest_dangling');
    raise exception 'ROLLBACK_SELFTEST:%', public.verify_dangling_scrape_run('selftest', 'zz_selftest_dangling');
  exception when raise_exception then
    if sqlerrm <> 'ROLLBACK_SELFTEST:t' then
      raise exception 'dangling remediation self-test failed: %', sqlerrm;
    end if;
  end;
end
$check$;
